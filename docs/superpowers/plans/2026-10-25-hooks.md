# Hooks 生命周期钩子 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 四事件生命周期钩子（PreToolUse 拦截/PostToolUse 追加/Stop/SessionStart 通知），两层 hooks.json + 项目层 sha256 信任 + `/hooks` 面板。

**Architecture:** 新增 `agent.hooks` 包（模型/加载器/信任注册表/执行器），Pre/Post 挂进 `ToolEventCallback` 第 4 可选参（null=停用，MCP 装饰点零改动），Stop 经 `bindHooks` 异步提交，SessionStart 在 `assemble`；registry 是活单实例（信任态 volatile，`t` 即时生效）。

**Tech Stack:** Java 17、JUnit 5、Jackson 3（`tools.jackson`）、`ProcessBuilder`、reactor。

**Spec:** `docs/superpowers/specs/2026-10-25-hooks-design.md`（执行语义表/挂载接线/残余风险是行为权威；执行者先读 spec）。

```java
package io.github.javaside.springai.codetui.agent.hooks;

/** 四个生命周期事件（配置键用官方拼写，fromConfigKey 映射；未知键返回 null 由 loader 跳过）。 */
public enum HookEvent {
    PRE_TOOL_USE("PreToolUse"), POST_TOOL_USE("PostToolUse"), STOP("Stop"), SESSION_START("SessionStart");

    private final String configKey;
    HookEvent(String configKey) { this.configKey = configKey; }
    public String configKey() { return configKey; }

    static HookEvent fromConfigKey(String key) {
        for (HookEvent e : values()) {
            if (e.configKey.equals(key)) return e;
        }
        return null;
    }
}
```

```java
package io.github.javaside.springai.codetui.agent.hooks;

import java.util.regex.Pattern;

/**
 * 一条钩子定义（loader 产出、不可变）。matcher 是工具名正则（空串=全部）；
 * <b>上界 120s 刻意远小于流空闲默认 300s</b>（spec 事实 #13）——钩子在工具线程同步静默，
 * 逼近空闲判定会自造「流重试→工具重跑→钩子再跑」循环。
 */
public record HookDefinition(String matcher, String command, int timeoutSeconds,
                             HookEvent event, HookSource source) {

    /** STOP/SESSION_START 无工具名概念（matcher 被忽略），恒匹配；工具事件按正则整段匹配。 */
    public boolean matchesTool(String toolName) {
        if (event == HookEvent.STOP || event == HookEvent.SESSION_START) return true;
        if (matcher == null || matcher.isEmpty()) return true;
        if (toolName == null) return false;   // 工具事件给了 matcher 却拿不到工具名：不匹配比误全跑安全
        return Pattern.matches(matcher, toolName);
    }
}
```

```java
package io.github.javaside.springai.codetui.agent.hooks;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import tools.jackson.databind.ObjectMapper;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.regex.Pattern;
import java.util.regex.PatternSyntaxException;

/**
 * 两层 hooks.json 加载（契约照 McpConfigLoader：文件缺失/JSON 非法/坏条目 → WARN 跳过，绝不抛）。
 * 合并语义是 concat（用户层在前、全部执行）——钩子无名字可覆盖，同事件全跑。
 */
public final class HookConfigLoader {

    private static final Logger log = LoggerFactory.getLogger(HookConfigLoader.class);
    private static final ObjectMapper MAPPER = new ObjectMapper();
    static final int DEFAULT_TIMEOUT = 60;
    static final int MAX_TIMEOUT = 120;

    record HookDto(String matcher, String command, Integer timeout) {}
    record HooksFileDto(Map<String, List<HookDto>> hooks) {}

    private HookConfigLoader() {}

    /** 生产入口：~/.codetui/hooks.json + <root>/.codetui/hooks.json。 */
    public static List<HookDefinition> load(Path root) {
        return load(Path.of(System.getProperty("user.home")).resolve(".codetui").resolve("hooks.json"),
                root.resolve(".codetui").resolve("hooks.json"));
    }

    /** 可测入口（测试不碰 user.home）。 */
    public static List<HookDefinition> load(Path userFile, Path projectFile) {
        List<HookDefinition> out = new ArrayList<>();
        out.addAll(parse(userFile, HookSource.USER));
        out.addAll(parse(projectFile, HookSource.PROJECT));
        return out;
    }

    private static List<HookDefinition> parse(Path file, HookSource source) {
        if (!Files.isRegularFile(file)) {
            return List.of();
        }
        HooksFileDto dto;
        try {
            dto = MAPPER.readValue(Files.readString(file), HooksFileDto.class);
        } catch (RuntimeException | java.io.IOException e) {
            log.warn("跳过无法解析的 hooks 配置 {}：{}", file, e.toString());
            return List.of();
        }
        if (dto == null || dto.hooks() == null) {
            return List.of();
        }
        List<HookDefinition> out = new ArrayList<>();
        for (var entry : dto.hooks().entrySet()) {
            HookEvent event = HookEvent.fromConfigKey(entry.getKey());
            if (event == null) {
                log.warn("跳过未知事件键 {}（支持：PreToolUse/PostToolUse/Stop/SessionStart）", entry.getKey());
                continue;
            }
            if (entry.getValue() == null) continue;
            for (HookDto h : entry.getValue()) {
                HookDefinition def = toDefinition(h, event, source, file);
                if (def != null) out.add(def);
            }
        }
        return out;
    }

    private static HookDefinition toDefinition(HookDto h, HookEvent event, HookSource source, Path file) {
        if (h == null || h.command() == null || h.command().isBlank()) {
            log.warn("跳过缺 command 的钩子（{} / {}）", file, event.configKey());
            return null;
        }
        String matcher = (h.matcher() == null || h.matcher().isBlank()) ? "" : h.matcher();
        if (!matcher.isEmpty()) {
            try {
                Pattern.compile(matcher);   // 只验证可编译；matchesTool 每次现编译（钩子低频，不值得缓存）
            } catch (PatternSyntaxException e) {
                log.warn("跳过 matcher 非法的钩子（{}）：{}", matcher, e.toString());
                return null;
            }
        }
        return new HookDefinition(matcher, h.command().strip(), clampTimeout(h.timeout()), event, source);
    }

    private static int clampTimeout(Integer configured) {
        if (configured == null) return DEFAULT_TIMEOUT;
        if (configured < 1) { log.warn("timeout={} 非法，回退默认 {}", configured, DEFAULT_TIMEOUT); return DEFAULT_TIMEOUT; }
        return Math.min(configured, MAX_TIMEOUT);
    }
}
```

```java
    void timeoutKills(@TempDir Path root) {

        assertTrue(r.timedOut());
        assertEquals(-1, r.exitCode(), "超时无退出码");
        assertTrue((System.nanoTime() - t0) < 20_000_000_000L, "1s 超时应在远小于 30s 内返回");
    }

    @Test
    void missingCommandNonThrowing(@TempDir Path root) {
        HookResult r = HookExecutor.execute(def("definitely-not-a-command-xyz", 10), "{}", root);
        assertEquals(-1, r.exitCode(), "启动失败以 -1 表达，不抛");
        assertTrue(!r.stderr().isEmpty());
        // 同 cwd：工作目录确为 root（钩子里的相对路径以此解析）
        assertEquals("ok", HookExecutor.execute(def("test -d . && echo ok", 10), "{}", root).stdout().strip());
```

```java
package io.github.javaside.springai.codetui.agent.hooks;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.io.IOException;
import java.io.OutputStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.concurrent.TimeUnit;

/**
 * 钩子进程执行：{@code sh -c command}、cwd=root、stdin 喂 JSON payload。
 *
 * <p><b>读线程的必要性</b>：若先 waitFor 再读输出，钩子写满管道缓冲（通常 64KB）后会阻塞在
 * write 上、永远退不出——prettier 这类多输出钩子会把自身卡到超时被杀。故先起两个守护读线程
 * 有界吸走输出，再等进程。
 *
 * <p><b>永不抛</b>：启动失败/IO 异常一律 {@code HookResult(-1,…)}——钩子坏不该打断工具调用
 * （工具容错白名单的老账）。
 */
public final class HookExecutor {

    private static final Logger log = LoggerFactory.getLogger(HookExecutor.class);
    private static final int OUTPUT_CAP = 64 * 1024;   // 每流捕获上限：payload 回传另有 2000 字截断

    public record HookResult(int exitCode, String stdout, String stderr, boolean timedOut) {}

    private HookExecutor() {}

    public static HookResult execute(HookDefinition def, String payloadJson, Path root) {
        Process p;
        try {
            p = new ProcessBuilder("sh", "-c", def.command())
                    .directory(root.toFile())
                    .start();
        } catch (IOException e) {
            log.warn("钩子进程启动失败（{}）：{}", def.command(), e.toString());
            return new HookResult(-1, "", "启动失败：" + e, false);
        }
        StringBuilder out = new StringBuilder();
        StringBuilder err = new StringBuilder();
        Thread tOut = drain(p.getInputStream(), out);
        Thread tErr = drain(p.getErrorStream(), err);
        try (OutputStream stdin = p.getOutputStream()) {
            stdin.write(payloadJson.getBytes(StandardCharsets.UTF_8));
        } catch (IOException e) {
            log.debug("写钩子 stdin 失败（钩子可能已提前退出）：{}", e.toString());
        }
        boolean finished;
        try {
            finished = p.waitFor(def.timeoutSeconds(), TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            p.destroyForcibly();
            return new HookResult(-1, out.toString(), "等待被中断", false);
        }
        if (!finished) {
            p.destroyForcibly();
            return new HookResult(-1, out.toString(), "超时（>" + def.timeoutSeconds() + "s）", true);
        }
        joinQuietly(tOut);
        joinQuietly(tErr);
        return new HookResult(p.exitValue(), out.toString(), err.toString(), false);
    }

    private static Thread drain(java.io.InputStream in, StringBuilder sb) {
        Thread t = new Thread(() -> {
            try (in) {
                byte[] buf = new byte[4096];
                int n;
                while (sb.length() < OUTPUT_CAP && (n = in.read(buf)) >= 0) {
                    sb.append(new String(buf, 0, n, StandardCharsets.UTF_8));
                }
            } catch (IOException ignored) {
                // 进程退出后 read 抛 EOF/Broken pipe 属正常收尾
            }
        }, "hook-drain");
        t.setDaemon(true);
        t.start();
        return t;
    }

    private static void joinQuietly(Thread t) {
        try {
            t.join(1000);   // 有界 join：读线程被喂满 64KB 后自会退，1s 兜底防挂测试
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }
}
```

```java
package io.github.javaside.springai.codetui.agent.hooks;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import tools.jackson.databind.ObjectMapper;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.security.MessageDigest;
import java.time.Instant;
import java.util.ArrayList;
import java.util.HexFormat;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.function.Consumer;

/**
 * 钩子注册表——<b>活的单实例</b>（spec 定案）：钩子表装配期不可变；项目层信任态 volatile
 * （UI 线程 {@link #trustProjectLayer} 写 / 工具与 reactive 线程 {@link #hooksFor} 读），
 * 信任翻转对装饰链里同一个实例即时生效。
 *
 * <p><b>信任模型</b>：项目层 hooks.json 的 sha256 记在 ~/.codetui/hooks-trust.json（用户空间，
 * 按项目绝对路径）；内容任何变更即失信任。残余风险（agent 自写信任文件）见 spec——已声明接受。
 */
public final class HookRegistry {

    private static final Logger log = LoggerFactory.getLogger(HookRegistry.class);
    private static final ObjectMapper MAPPER = new ObjectMapper();
    private static final int REPLY_CAP = 2000;      // Pre stderr / Post stdout 回传上限（spec）
    private static final int PAYLOAD_OUTPUT_CAP = 8 * 1024;   // Post payload 里 toolOutput 截断

    public record PreResult(boolean blocked, String message) {
        static final PreResult ALLOW = new PreResult(false, null);
    }

    record TrustFileDto(Map<String, String> projects) {}

    private final Path root;
    private final List<HookDefinition> hooks;          // 装配期不可变
    private final Path projectFile;
    private final Path trustFile;
    private final String projectRootKey;               // 项目根绝对路径（信任键）
    private final String projectHash;                  // 项目层内容 hash（无文件则 null）
    /** volatile：UI 线程写（trustProjectLayer）、工具/reactive 线程读（hooksFor）。 */
    private volatile boolean projectTrusted;
    /** 面板 mtime 可疑提示用（未信任则 null；t 翻转时由 UI 线程写，volatile 同上）。 */
    private volatile Instant trustedAt;
    private final ExecutorService notifyExecutor =
            Executors.newSingleThreadExecutor(r -> {
                Thread t = new Thread(r, "hook-notify");
                t.setDaemon(true);
                return t;
            });

    private HookRegistry(Path root, List<HookDefinition> hooks, Path projectFile,
                         Path trustFile, boolean projectTrusted, Instant trustedAt) {
        this.root = root;
        this.hooks = List.copyOf(hooks);
        this.projectFile = projectFile;
        this.trustFile = trustFile;
        this.projectRootKey = root.toAbsolutePath().toString();
        this.projectHash = sha256OfFile(projectFile);
        this.projectTrusted = projectTrusted;
        this.trustedAt = trustedAt;
    }

    /** 生产入口：标准两层路径 + 用户空间信任文件。 */
    public static HookRegistry load(Path root) {
        return load(root,
                Path.of(System.getProperty("user.home")).resolve(".codetui").resolve("hooks.json"),
                root.resolve(".codetui").resolve("hooks.json"),
                Path.of(System.getProperty("user.home")).resolve(".codetui").resolve("hooks-trust.json"));
    }

    /** 可测入口（不碰 user.home）。 */
    public static HookRegistry load(Path root, Path userFile, Path projectFile, Path trustFile) {
        List<HookDefinition> all = HookConfigLoader.load(userFile, projectFile);
        boolean trusted = false;
        Instant trustedAt = null;
        String currentHash = sha256OfFile(projectFile);
        if (currentHash != null) {
            TrustFileDto dto = readTrust(trustFile);
            String recorded = (dto == null || dto.projects() == null) ? null
                    : dto.projects().get(root.toAbsolutePath().toString());
            if (currentHash.equals(recorded)) {
                trusted = true;
                trustedAt = mtime(projectFile);
            }
        }
        return new HookRegistry(root, all, projectFile, trustFile, trusted, trustedAt);
    }

    // ── 查询 ────────────────────────────────────────────────

    public boolean projectLayerUntrusted() {
        return projectHash != null && !projectTrusted;
    }

    /** 项目层 hooks.json 的 mtime 晚于信任时刻 → ⚠ 可疑（残余风险的廉价提示）；可信或无配置返回空串。 */
    public String projectFileMtimeNotice() {
        if (!projectLayerUntrusted() && projectTrusted && trustedAt != null) {
            Instant mtime = mtime(projectFile);
            if (mtime != null && mtime.isAfter(trustedAt)) {
                return "⚠ 项目层 hooks.json 在信任后被修改过，建议复查（/hooks 查看内容）";
            }
        }
        return "";
    }

    /** 每次调用按当前信任态过滤（信任翻转即时生效的支点）。 */
    public List<HookDefinition> hooksFor(HookEvent event, String toolName) {
        List<HookDefinition> out = new ArrayList<>();
        for (HookDefinition h : hooks) {
            if (h.event() != event) continue;
            if (h.source() == HookSource.PROJECT && !projectTrusted) continue;
            if (h.matchesTool(toolName)) out.add(h);
        }
        return out;
    }

    // ── 信任写入 ────────────────────────────────────────────

    /** 面板 t 键：原子写信任文件 + 就地翻转内存态（同一实例即时生效）。 */
    public synchronized void trustProjectLayer() {
        String hash = sha256OfFile(projectFile);
        if (hash == null) return;   // 项目层文件没了：无事可信任
        TrustFileDto dto = readTrust(trustFile);
        Map<String, String> projects = (dto == null || dto.projects() == null)
                ? new LinkedHashMap<>() : new LinkedHashMap<>(dto.projects());
        projects.put(projectRootKey, hash);
        atomicWriteTrust(new TrustFileDto(projects));
        this.projectTrusted = true;
        this.trustedAt = Instant.now();
    }

    // ── 事件入口（Pre/Post 给 ToolEventCallback；Notify 给 Stop/SessionStart）────

    /** PreToolUse：逐个同步跑，首个 exit 2 胜出；非 2 失败 WARN 继续。永不抛。 */
    public PreResult runPre(String toolName, String toolInput, String sessionId, long turnId, String taskId) {
        for (HookDefinition h : hooksFor(HookEvent.PRE_TOOL_USE, toolName)) {
            HookExecutor.HookResult r = HookExecutor.execute(h,
                    payload(h, toolName, toolInput, null, sessionId, turnId, taskId), root);
            if (r.exitCode() == 2) {
                return new PreResult(true, "⛔ PreToolUse 钩子拦截（" + h.command() + "）："
                        + cap(r.stderr()));
            }
            if (r.exitCode() != 0) {
                log.warn("PreToolUse 钩子非阻断失败（{}）：exit={} stderr={}", h.command(), r.exitCode(), cap(r.stderr()));
            }
        }
        return PreResult.ALLOW;
    }

    /** PostToolUse：exit 0 且 stdout 非空者追加（多钩子依次）；失败不追加。永不抛。 */
    public String runPost(String toolName, String toolInput, String toolOutput,
                          String sessionId, long turnId, String taskId) {
        StringBuilder append = null;
        for (HookDefinition h : hooksFor(HookEvent.POST_TOOL_USE, toolName)) {
            HookExecutor.HookResult r = HookExecutor.execute(h,
                    payload(h, toolName, toolInput, toolOutput, sessionId, turnId, taskId), root);
            if (r.exitCode() == 0 && !r.stdout().isBlank()) {
                if (append == null) append = new StringBuilder();
                append.append("\n\n[PostToolUse 钩子输出]\n").append(cap(r.stdout()));
            } else if (r.exitCode() != 0) {
                log.warn("PostToolUse 钩子失败不追加（{}）：exit={} stderr={}", h.command(), r.exitCode(), cap(r.stderr()));
            }
        }
        return append == null ? null : append.toString();
    }

    /** 通知型同步执行（SessionStart：assemble 里跑，阻塞可接受）。 */
    public void runNotify(HookEvent event, Consumer<String> notice) {
        for (HookDefinition h : hooksFor(event, null)) {
            HookExecutor.HookResult r = HookExecutor.execute(h,
                    payload(h, null, null, null, null, -1L, null), root);
            if (r.exitCode() != 0) {
                notice.accept("⚠ " + event.configKey() + " 钩子失败（" + h.command() + "）："
                        + firstLine(r.stderr()));
            }
        }
    }

    /** 通知型异步执行（Stop：不阻塞 reactive 线程——spec 定案，次序损失已接受）。 */
    public void runNotifyAsync(HookEvent event, Consumer<String> notice) {
        List<HookDefinition> snapshot = hooksFor(event, null);
        if (snapshot.isEmpty()) return;
        // 捕获快照而非 registry 引用：信任态在提交时已判定（异步窗口内翻转不再影响本批）
        for (HookDefinition h : snapshot) {
            notifyExecutor.execute(() -> {
                try {
                    HookExecutor.HookResult r = HookExecutor.execute(h,
                            payload(h, null, null, null, null, -1L, null), root);
                    if (r.exitCode() != 0) {
                        notice.accept("⚠ " + event.configKey() + " 钩子失败（" + h.command() + "）："
                                + firstLine(r.stderr()));
                    }
                } catch (RuntimeException e) {   // 防御：钩子链任何异常不许杀 notify 线程
                    log.warn("Stop 钩子执行异常（{}）：{}", h.command(), e.toString());
                }
            });
        }
    }

    // ── 内部 ────────────────────────────────────────────────

    private String payload(HookDefinition h, String toolName, String toolInput, String toolOutput,
                           String sessionId, Long turnId, String taskId) {
        Map<String, Object> m = new LinkedHashMap<>();
        m.put("event", h.event().configKey());
        if (toolName != null) m.put("toolName", toolName);
        if (toolInput != null) m.put("toolInput", toolInput);
        if (toolOutput != null) m.put("toolOutput", capTo(toolOutput, PAYLOAD_OUTPUT_CAP));
        m.put("sessionId", sessionId == null ? "" : sessionId);
        m.put("turnId", turnId == null ? -1L : turnId);
        m.put("taskId", taskId == null ? "" : taskId);
        m.put("root", root.toString());
        return MAPPER.writeValueAsString(m);   // Jackson 3 writeValueAsString 不抛受检异常
    }

    private static String cap(String s) { return capTo(s, REPLY_CAP); }

    private static String capTo(String s, int max) {
        if (s == null) return "";
        return s.length() <= max ? s : s.substring(0, max) + "…";
    }

    private static String firstLine(String s) {
        if (s == null || s.isBlank()) return "";
        String line = s.strip().split("\n", 2)[0];
        return line.length() > 120 ? line.substring(0, 120) + "…" : line;
    }

    private static String sha256OfFile(Path file) {
        if (!Files.isRegularFile(file)) return null;
        try {
            return HexFormat.of().formatHex(
                    MessageDigest.getInstance("SHA-256").digest(Files.readAllBytes(file)));
        } catch (Exception e) {
            log.warn("读取 {} 失败：{}", file, e.toString());
            return null;
        }
    }

    private static Instant mtime(Path file) {
        try {
            return Files.getLastModifiedTime(file).toInstant();
        } catch (Exception e) {
            return null;
        }
    }

    private static TrustFileDto readTrust(Path file) {
        if (!Files.isRegularFile(file)) return null;
        try {
            return MAPPER.readValue(Files.readString(file), TrustFileDto.class);
        } catch (RuntimeException | java.io.IOException e) {
            log.warn("信任文件不可解析 {}：{}", file, e.toString());
            return null;
        }
    }

    private void atomicWriteTrust(TrustFileDto dto) {
        try {
            Files.createDirectories(trustFile.getParent());
            Path tmp = trustFile.resolveSibling("hooks-trust.json.tmp");
            Files.writeString(tmp, MAPPER.writeValueAsString(dto));
            try {
                Files.move(tmp, trustFile, StandardCopyOption.ATOMIC_MOVE, StandardCopyOption.REPLACE_EXISTING);
            } catch (java.io.IOException atomicUnsupported) {
                Files.move(tmp, trustFile, StandardCopyOption.REPLACE_EXISTING);
            }
        } catch (RuntimeException | java.io.IOException e) {
            log.warn("写信任文件失败 {}：{}", trustFile, e.toString());
        }
    }
}
```

```java
    private void openHooksPanel() {
        hookIndex = 0;
        pickingHooks = true;
    }

    /** 键位：↑↓ 移动 · t 信任项目层（未信任时）· Esc 关闭；始终 HANDLED；空列表防除零。 */
    private EventResult onHooksPanelKey(KeyEvent k) {
        var registry = onSubmit.hookRegistry();
        if (k.isCancel()) { pickingHooks = false; return EventResult.HANDLED; }
        if (registry == null) return EventResult.HANDLED;
        int n = 0;
        for (var event : io.github.javaside.springai.codetui.agent.hooks.HookEvent.values()) {
            n += registry.hooksFor(event, null).size();
        }
        if (n == 0) return EventResult.HANDLED;
        hookIndex = clampIndex(hookIndex, n);
        if (k.code() == KeyCode.UP)   { hookIndex = (hookIndex - 1 + n) % n; return EventResult.HANDLED; }
        if (k.code() == KeyCode.DOWN) { hookIndex = (hookIndex + 1) % n; return EventResult.HANDLED; }
        if ((k.isChar('t') || k.isChar('T')) && registry.projectLayerUntrusted()) {
            registry.trustProjectLayer();
            state.setNotice("已信任项目层 hooks.json（本会话即时生效；内容再改需重新信任）");
            return EventResult.HANDLED;
        }
        return EventResult.HANDLED;
    }

    /** 按事件分组的钩子行 + 未信任标记 + mtime 可疑提示；registry null 走空态。 */
    private Element[] hooksPanelChildren() {
        if (!pickingHooks) return new Element[0];
        if (activePermission != null || activePlan != null || activeAsk != null) return new Element[0];
        var registry = onSubmit.hookRegistry();
        List<Element> els = new ArrayList<>();
        els.add(text("  ⚙ 钩子（↑↓ 选择 · t 信任项目层 · Esc 关闭）").style(PICK_TITLE));
        if (registry == null) {
            els.add(text("    （当前装配未启用钩子）").style(PICK_DESC));
            return els.toArray(new Element[0]);
        }
        boolean any = false;
        int idx = 0;
        for (var event : io.github.javaside.springai.codetui.agent.hooks.HookEvent.values()) {
            for (var h : registry.hooksFor(event, null)) {
                any = true;
                String untrusted = (h.source() == io.github.javaside.springai.codetui.agent.hooks.HookSource.PROJECT
                        && registry.projectLayerUntrusted()) ? " · ⚠未信任" : "";
                String line = "  " + event.configKey() + " · " + h.command()
                        + (h.matcher().isEmpty() ? "" : " · matcher=" + h.matcher())
                        + " · " + h.timeoutSeconds() + "s · " + h.source() + untrusted;
                els.add(text(line).style(idx == hookIndex ? PICK_SEL : PICK_DESC));
                idx++;
            }
        }
        if (!any) {
            els.add(text("    （暂无钩子）在 ~/.codetui/hooks.json 或 <root>/.codetui/hooks.json 配置；"
                    + "项目层首次使用需在此信任。").style(PICK_DESC));
        }
        String mtimeWarn = registry.projectFileMtimeNotice();
        if (!mtimeWarn.isEmpty()) els.add(text("  " + mtimeWarn).style(PICK_DESC));
        return els.toArray(new Element[0]);
    }

    String hooksStatusText() {
        String notice = state.notice();
        if (notice != null && !notice.isEmpty()) return notice + " · Esc 关闭";
        return "⚙ 钩子 · ↑↓ 选择 · t 信任 · Esc 关闭";
    }
```
